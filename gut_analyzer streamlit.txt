import streamlit as st

# 페이지 기본 설정
st.set_page_config(page_title="장 건강 성분 분석기", page_icon="🥤")

# 분석 로직 데이터
RESEARCH_DATA = {
    "스테비아": 5.6,
    "난소화성말토덱스트린": 1.0,
    "설탕": 1.0,
    "수크랄로스": 0.7,
    "아스파탐": 0.4,
}

# 제목
st.title("🥤 장 건강 성분 분석 시스템")
st.write("성분과 함량(g)을 입력하여 장 건강 지수를 분석해보세요.")

st.divider()

# 입력 폼 구성
options = ["선택 안 함"] + list(RESEARCH_DATA.keys())

col1, col2 = st.columns([2, 1])
with col1:
    ing1 = st.selectbox("성분 1", options, key="ing1")
with col2:
    amt1 = st.number_input("함량 1 (g)", min_value=0.0, step=0.1, key="amt1")

col3, col4 = st.columns([2, 1])
with col3:
    ing2 = st.selectbox("성분 2", options, key="ing2")
with col4:
    amt2 = st.number_input("함량 2 (g)", min_value=0.0, step=0.1, key="amt2")

col5, col6 = st.columns([2, 1])
with col5:
    ing3 = st.selectbox("성분 3", options, key="ing3")
with col6:
    amt3 = st.number_input("함량 3 (g)", min_value=0.0, step=0.1, key="amt3")

st.divider()

# 분석 버튼
if st.button("분석하기", type="primary", use_container_width=True):
    inputs = [(ing1, amt1), (ing2, amt2), (ing3, amt3)]
    total_impact = 1.0
    active_count = 0

    for name, amt in inputs:
        if name != "선택 안 함" and amt > 0:
            active_count += 1
            total_impact *= RESEARCH_DATA.get(name, 1.0) ** (amt / 10)

    if active_count == 0:
        st.warning("성분을 하나 이상 선택하고 함량을 입력해주세요.")
    else:
        score = round(100 * total_impact, 1)
        st.subheader(f"최종 지수: {score}점")

        if score >= 100:
            st.success("✅ 설탕 대비 우수함")
        else:
            st.error("🚨 설탕 대비 저해됨")
